Luau 01

Secure networking layer

A wrapper around RemoteEvents and RemoteFunctions that keeps the server in charge of everything the client sends.

  • Per-player, per-remote token-bucket rate limiting
  • Typed payload validation before any handler runs
  • Decaying strike score that kicks repeat offenders
  • Handlers isolated with xpcall so one error never takes the server down
  • Debug-only client checks that catch bad payloads before they ship
· · · trimmed · · ·
if IS_SERVER then
	local Server = {}
	Server.Violation = Signal.new()

	local folder = Instance.new("Folder")
	folder.Name = FOLDER_NAME
	local remotes: { [string]: Instance } = {}
	local handlers: { [string]: (...any) -> ...any } = {}
	local validators: { [string]: (...any) -> (boolean, string?) } = {}
	local buckets: { [Player]: { [string]: { tokens: number, last: number } } } = {}
	local strikes: { [Player]: { score: number, last: number } } = {}

	local function strike(player: Player, remoteName: string, kind: string, detail: string?)
		local now = os.clock()
		local entry = strikes[player]
		if not entry then
			entry = { score = 0, last = now }
			strikes[player] = entry
		end
		local decay = (now - entry.last) / Config.Security.ViolationDecaySeconds
		entry.score = math.max(0, entry.score - decay) + (STRIKE_WEIGHT[kind] or 1)
		entry.last = now
		local reason = if detail then kind .. ": " .. detail else kind
		if kind ~= "RateLimited" and kind ~= "Speed" then
			log:Warn(string.format("%s (%d) -> %s rejected: %s", player.Name, player.UserId, remoteName, reason))
		end
		Server.Violation:Fire(player, remoteName, reason)
		if entry.score >= Config.Security.MaxViolations and player.Parent then
			player:Kick("Disconnected: unexpected network activity.")
		end
	end
	Server.Strike = strike

	local function consumeToken(player: Player, remoteName: string, limit: { Count: number, Window: number }): boolean
		local perPlayer = buckets[player]
		if not perPlayer then
			if not player.Parent then
				return false
			end
			perPlayer = {}
			buckets[player] = perPlayer
		end
		local now = os.clock()
		local bucket = perPlayer[remoteName]
		if not bucket then
			bucket = { tokens = limit.Count, last = now }
			perPlayer[remoteName] = bucket
		end
		bucket.tokens = math.min(limit.Count, bucket.tokens + (now - bucket.last) * (limit.Count / limit.Window))
		bucket.last = now
		if bucket.tokens < 1 then
			return false
		end
		bucket.tokens -= 1
		return true
	end

	local function admit(player: Player, name: string, def: any, ...: any): boolean
		if typeof(player) ~= "Instance" or not player:IsA("Player") or player.Parent ~= Players then
			return false
		end
		if def.Direction == "ToClient" then
			strike(player, name, "WrongDirection")
			return false
		end
		if def.RateLimit and not consumeToken(player, name, def.RateLimit) then
			strike(player, name, "RateLimited")
			return false
		end
		local ok, err = validators[name](...)
		if not ok then
			strike(player, name, "InvalidPayload", err)
			return false
		end
		return true
	end

	local function runProtected(name: string, handler: (...any) -> ...any, ...: any)
		local ok, err = xpcall(handler, debug.traceback, ...)
		if not ok then
			log:Error(string.format("handler '%s' errored:\n%s", name, tostring(err)))
		end
	end

	for name, def in Definitions do
		local remote = Instance.new(CLASS_FOR_KIND[def.Kind])
		remote.Name = name
		remote.Parent = folder
		remotes[name] = remote
		validators[name] = Guard.tuple(def.Args or {})
		if def.Kind == "Function" then
			(remote :: RemoteFunction).OnServerInvoke = function(player: Player, ...)
				if not admit(player, name, def, ...) then
					return false, "Rejected"
				end
				local handler = handlers[name]
				if not handler then
					return false, "Unavailable"
				end
				local results = table.pack(xpcall(handler, debug.traceback, player, ...))
				if not results[1] then
					log:Error(string.format("function '%s' errored:\n%s", name, tostring(results[2])))
					return false, "InternalError"
				end
				return true, table.unpack(results, 2, results.n)
			end
		else
			(remote :: RemoteEvent).OnServerEvent:Connect(function(player: Player, ...)
				if not admit(player, name, def, ...) then
					return
				end
				local handler = handlers[name]
				if handler then
					task.spawn(runProtected, name, handler, player, ...)
				end
			end)
		end
	end

	local existing = ReplicatedStorage:FindFirstChild(FOLDER_NAME)
	if existing then
		existing:Destroy()
	end
	folder.Parent = ReplicatedStorage

	Players.PlayerRemoving:Connect(function(player)
		buckets[player] = nil
		strikes[player] = nil
	end)

	local function assertOutbound(name: string)
		local def = getDefinition(name)
		assert(def.Kind ~= "Function", "[Net] Functions cannot be fired from the server")
		assert(def.Direction ~= "ToServer", "[Net] '" .. name .. "' is ToServer only")
	end

	function Server.Connect(name: string, handler: (player: Player, ...any) -> ())
		local def = getDefinition(name)
		assert(def.Kind ~= "Function", "[Net] use SetHandler for Functions")
		assert(handlers[name] == nil, "[Net] '" .. name .. "' already has a handler")
		handlers[name] = handler
	end

	function Server.SetHandler(name: string, handler: (player: Player, ...any) -> ...any)
		local def = getDefinition(name)
		assert(def.Kind == "Function", "[Net] SetHandler is only for Functions")
		assert(handlers[name] == nil, "[Net] '" .. name .. "' already has a handler")
		handlers[name] = handler
	end

	function Server.Fire(name: string, player: Player, ...: any)
		assertOutbound(name);
		(remotes[name] :: RemoteEvent):FireClient(player, ...)
	end

	function Server.FireAll(name: string, ...: any)
		assertOutbound(name);
		(remotes[name] :: RemoteEvent):FireAllClients(...)
	end

	function Server.FireList(name: string, list: { Player }, ...: any)
		assertOutbound(name)
		local remote = remotes[name] :: RemoteEvent
		for _, player in list do
			remote:FireClient(player, ...)
		end
	end

	Net.Server = Server
else
	local Client = {}
	local folder: Folder? = nil
	local cache: { [string]: Instance } = {}

	local function getRemote(name: string): Instance
		local remote = cache[name]
		if remote then
			return remote
		end
		if not folder then
			folder = ReplicatedStorage:WaitForChild(FOLDER_NAME, 30) :: Folder
			assert(folder, "[Net] remotes folder never replicated")
		end
		remote = (folder :: Folder):WaitForChild(name, 30)
		assert(remote, "[Net] remote '" .. name .. "' never replicated")
		cache[name] = remote
		return remote
	end

	local function devValidate(name: string, def: any, ...: any)
		if Config.Debug and def.Args then
			local ok, err = Guard.tuple(def.Args)(...)
			if not ok then
				error(string.format("[Net] '%s' would be rejected: %s", name, err or "?"), 3)
			end
		end
	end

	function Client.Fire(name: string, ...: any)
		local def = getDefinition(name)
		assert(def.Kind ~= "Function", "[Net] use Invoke for Functions")
		devValidate(name, def, ...);
		(getRemote(name) :: RemoteEvent):FireServer(...)
	end

	function Client.Invoke(name: string, ...: any): (boolean, ...any)
		local def = getDefinition(name)
		assert(def.Kind == "Function", "[Net] Invoke is only for Functions")
		devValidate(name, def, ...)
		local remote = getRemote(name) :: RemoteFunction
		local results = table.pack(pcall(remote.InvokeServer, remote, ...))
		if not results[1] then
			return false, "NetworkError"
		end
		return table.unpack(results, 2, results.n)
	end

	function Client.Connect(name: string, handler: (...any) -> ()): RBXScriptConnection
		local def = getDefinition(name)
		assert(def.Kind ~= "Function", "[Net] clients cannot handle Functions")
		return (getRemote(name) :: RemoteEvent).OnClientEvent:Connect(handler)
	end

	Net.Client = Client
end
Luau 02

Analytic hazard collision

Tests a player position against moving pushers and rotating sweeper arms at any time t, and returns the knockback vector.

  • Pure math — no physics parts or touch events needed
  • Deterministic, so the server can verify hits the client reports
  • Pushes the player in the direction the hazard is moving
· · · trimmed · · ·
function HazardService:Test(pos: Vector3, t: number): Vector3?
	if pos.Y > JUMP_CLEAR then
		return nil
	end
	for _, p in pushers do
		local arg = p.Phase + p.Speed * t
		local bx = p.Center.X + p.Amp * math.sin(arg)
		if math.abs(pos.Z - p.Center.Z) <= p.Half.Z + 1.3 and math.abs(pos.X - bx) <= p.Half.X + 1.3 then
			local vel = math.cos(arg)
			local side = if math.abs(vel) > 0.2 then math.sign(vel) else math.sign(pos.X - bx)
			return Vector3.new(side * 68, 34, 0)
		end
	end
	for _, s in sweepers do
		local v = Vector3.new(pos.X - s.Center.X, 0, pos.Z - s.Center.Z)
		if v.Magnitude <= s.Radius + 2 then
			local a = HazardService.Angle(s, t)
			local dir = Vector3.new(math.cos(a), 0, math.sin(a))
			local along = v:Dot(dir)
			if (v - dir * along).Magnitude <= ARM_HALF + 1 then
				local tangent = Vector3.new(-dir.Z, 0, dir.X) * math.sign(s.Speed) * math.sign(along)
				return tangent * 75 + Vector3.new(0, 38, 0)
			end
		end
	end
	return nil
end
C# 03

Sub-stepped axis movement

Moves a body along one axis in small steps so fast objects can't tunnel through walls.

  • Step count scales with speed
  • Sets wall, floor and ceiling contact flags
  • Raises a collision event with the hit normal
· · · trimmed · · ·
void MoveAxis(float amount, bool horizontal, float maxStep)
    {
        if (amount == 0f) return;
        var e = Owner;
        int steps = Math.Max(1, (int)MathF.Ceiling(MathF.Abs(amount) / maxStep));
        float step = amount / steps;

        for (int i = 0; i < steps; i++)
        {
            var prevBounds = Bounds;
            if (horizontal) e.Position.X += step; else e.Position.Y += step;
            if (Resolve(horizontal, step, prevBounds))
            {
                float dir = MathF.Sign(step);
                if (horizontal)
                {
                    Velocity.X = 0f;
                    if (dir > 0) OnWallRight = true; else OnWallLeft = true;
                    Collided?.Invoke(new Vector2(dir, 0));
                }
                else
                {
                    Velocity.Y = 0f;
                    if (dir > 0) OnFloor = true; else OnCeiling = true;
                    Collided?.Invoke(new Vector2(0, dir));
                }
                return;
            }
        }
    }
C# 04

Tilemap probe with one-way platforms

Checks whether a body would overlap solid tiles if moved by an offset, including standing on one-way platforms.

  • Works across every solid tilemap in the scene
  • Supports drop-through on one-way tiles
  • Tolerance check so bodies resting exactly on a tile stay grounded
· · · trimmed · · ·
 public bool Probe(Vector2 offset)
    {
        var e = Owner;
        if (e.Scene == null) return false;
        var b = Bounds;
        var moved = new Rectangle(b.X + offset.X, b.Y + offset.Y, b.Width, b.Height);
        foreach (var m in e.Scene.All<Tilemap>())
        {
            if (!m.Solid) continue;
            if (m.Overlaps(moved)) return true;
            if (offset.Y > 0 && m.OneWayTiles != null && !DropThrough)
            {
                // Standing exactly on top of a one-way tile?
                var (x0, _, x1, _) = m.CellsIn(moved);
                var (_, cy) = m.WorldToCell(new Vector2(b.X, b.Y + b.Height + 0.01f));
                for (int x = x0; x <= x1; x++)
                    if (m.IsOneWayTile(m.Get(x, cy)) && MathF.Abs(m.CellRect(x, cy).Y - (b.Y + b.Height)) < 0.05f)
                        return true;
            }
        }
        return false;
    }
JavaScript 05

Real-time WebGL render loop

The loop behind the animated 3D planet on horizonrblx.com, built with three.js.

  • Damped, pointer-driven camera drift
  • Pauses rendering when off-screen or in a background tab
  • Responsive framing with ResizeObserver
  • Respects reduced-motion settings and cleans up GPU resources
· · · trimmed · · ·
 const target = { x: 0, y: 0 };
    const current = { x: 0, y: 0 };

    const onPointer = (event) => {
      const rect = stage.getBoundingClientRect();
      if (!rect.width) return;
      target.x = THREE.MathUtils.clamp(((event.clientX - rect.left) / rect.width) * 2 - 1, -1.2, 1.2);
      target.y = THREE.MathUtils.clamp(((event.clientY - rect.top) / rect.height) * 2 - 1, -1.2, 1.2);
    };
    window.addEventListener("pointermove", onPointer, { passive: true });
    stage.addEventListener("pointerleave", () => { target.x = 0; target.y = 0; });

    const resize = () => {
      const rect = mount.getBoundingClientRect();
      if (rect.width < 2 || rect.height < 2) return;
      renderer.setSize(rect.width, rect.height, false);
      camera.aspect = rect.width / rect.height;
      /* keep the whole ring in frame on narrow cards */
      CAM_BASE.z = 10.4 * Math.max(1, 1.25 / camera.aspect);
      camera.updateProjectionMatrix();
    };

    const ro = new ResizeObserver(resize);
    ro.observe(mount);
    observers.push(ro);
    resize();

    const io = new IntersectionObserver(([entry]) => { visible = Boolean(entry?.isIntersecting); }, { threshold: 0.01 });
    io.observe(stage);
    observers.push(io);

    let last = performance.now();
    let time = 0;
    const moonOrbit = new THREE.Vector3();
    let shown = false;

    if (reducedMotion) moon.position.set(-2.6, 1.1, 1.2);

    const animate = () => {
      if (!running) return;
      frameId = requestAnimationFrame(animate);

      const now = performance.now();
      const dt = Math.min((now - last) / 1000, 0.05);
      last = now;

      if (!visible || document.hidden) return;

      time += dt;

      if (!reducedMotion) {
        planet.rotation.y += dt * 0.09;
        planetUniforms.uTime.value = time;
        starMat.uniforms.uTime.value = time;

        const a = time * 0.32;
        moonOrbit.set(Math.cos(a) * 3.35, Math.sin(a * 0.5) * 0.25, Math.sin(a) * 3.35);
        moon.position.copy(moonOrbit.applyQuaternion(system.quaternion));
      }

      /* damped, small camera drift — the system itself never tilts */
      current.x += (target.x - current.x) * Math.min(1, dt * 3);
      current.y += (target.y - current.y) * Math.min(1, dt * 3);
      camera.position.set(
        CAM_BASE.x + current.x * 0.9,
        CAM_BASE.y - current.y * 0.6,
        CAM_BASE.z
      );
      camera.lookAt(0, 0.3, 0);

      renderer.render(scene, camera);

      if (!shown) {
        shown = true;
        mount.classList.add("ready");
      }
    };

    animate();

    window.addEventListener("pagehide", () => {
      running = false;
      cancelAnimationFrame(frameId);
      observers.forEach((o) => o.disconnect());
      renderer.dispose();
    }, { once: true });
  } catch (error) {
    fail(error);
  }
